💡 USAGE EXAMPLES
─────────────────

// Initialize (in MainActivity.onCreate)
val debugManager = DebugFeaturesManager.getInstance(this)
debugManager.initialize(rootView, deviceIdText)

// Log messages
debugManager.logInfo("User logged in")
debugManager.logWarning("Memory usage high")
debugManager.logError("Sync failed", exception)
debugManager.logSuccess("✓ Data synced")
debugManager.logPerformance("Query took 250ms")
debugManager.logNetwork("Uploaded 5MB")
debugManager.logThermal("Device heating")

// Record sync events
debugManager.setSyncInProgress(true)
debugManager.recordSync(itemCount = 150, status = "Success")
debugManager.setSyncInProgress(false)

// Access performance metrics
val metrics = debugManager.getPerformanceMetrics()
println("Memory: ${metrics.memoryUsage.usedPercentage}%")
println("Battery: ${metrics.batteryStatus.level}%")
